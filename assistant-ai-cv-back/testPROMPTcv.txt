import { Injectable, Logger } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { InjectModel } from '@nestjs/mongoose';
import { OpenAI } from 'openai';
import { OpenAIUsage } from './schema/openai.usage.schema';
import { Model } from 'mongoose';

@Injectable()
export class OpenAiService {
  logger = new Logger(OpenAiService.name);
  client: OpenAI;
  constructor(
    private configService: ConfigService,
    @InjectModel(OpenAIUsage.name)
    private readonly openAIUsageModel: Model<OpenAIUsage>,
  ) {
    this.client = new OpenAI({
      apiKey: this.configService.getOrThrow<string>('OPENAI_API_KEY'),
    });
  }

  private sanitizeText(text: string): string {
    return text
      .replace(/[\x00-\x08\x0B\x0C\x0E-\x1F]/g, '') // Remove control characters
      .replace(/\\/g, '\\\\') // Escape backslashes
      .replace(/"/g, '\\"') // Escape double quotes
      .replace(/\n/g, '\\n') // Escape newlines
      .replace(/\r/g, '\\r') // Escape carriage returns
      .replace(/\t/g, '\\t'); // Escape tabs
  }

  private isValidJson(str: string): { valid: boolean; cleaned?: string } {
    try {
      JSON.parse(str);
      return { valid: true, cleaned: str };
      // eslint-disable-next-line @typescript-eslint/no-unused-vars
    } catch (error) {
      const cleaned = str
        .replace(/[\x00-\x08\x0B\x0C\x0E-\x1F]/g, '') // Remove invalid control characters
        .trim()
        .replace(/^```json\n|```$/g, '') // Remove markdown
        .replace(/^\s*[\{\[]/, '{') // Ensure starts with { or [
        .replace(/[\}\]]\s*$/, '}'); // Ensure ends with } or ]
      try {
        JSON.parse(cleaned);
        return { valid: true, cleaned };
      } catch (cleanError) {
        this.logger.error('Failed to parse cleaned JSON', cleanError);
        return { valid: false };
      }
    }
  }

  async processCV(text: string): Promise<any> {
    try {
      const sanitizedText = this.sanitizeText(text);
      this.logger.log('process start');
      const startTime = Date.now();
      const response = await this.client.chat.completions.create({
        model: 'gpt-4o-mini',
        messages: [
          {
            role: 'system',
            content:
              'You are a resume parsing AI that extracts structured data in JSON format. Return JSON only, without markdown formatting or code blocks. Pay special attention to extracting meaningful keywords and projects from the CV.',
          },
          { role: 'user', content: this.getPromptText(sanitizedText) },
        ],
        temperature: 0.2, // Low temperature for consistency
      });
      const endTime = Date.now();
      const timeToProcess = endTime - startTime; // Calculate time taken to process

      this.logger.log('process end');

      const rawContent = response.choices[0].message.content;

      const jsonValidation = this.isValidJson(rawContent);

      if (!jsonValidation.valid) {
        this.logger.error('Invalid JSON response from OpenAI', rawContent);
        return {
          valid: false,
          validation_message: 'Invalid JSON response from OpenAI',
        };
      }

      const schema = new this.openAIUsageModel({
        model: 'gpt-4o-mini',
        usage: response.usage,
        time_to_process: timeToProcess,
      });

      await schema.save();

      const extractedJSON =
        jsonValidation.cleaned || response.choices[0].message.content;
      const parsedResponse = JSON.parse(extractedJSON);

      // Ensure keywords are extracted if none were returned
      if (!parsedResponse.keywords || parsedResponse.keywords.length === 0) {
        this.logger.log(
          'No keywords found in initial response, extracting keywords separately',
        );
        // Extract keywords in a separate call if needed
        const keywordsResult = await this.extractKeywords(sanitizedText);
        parsedResponse.keywords = keywordsResult.keywords || [];
      }

      // New: Ensure projects are extracted if none were returned
      if (!parsedResponse.projects || parsedResponse.projects.length === 0) {
        this.logger.log(
          'No projects found in initial response, extracting projects separately',
        );
        // Extract projects in a separate call
        const projectsResult = await this.extractProjects(sanitizedText);
        parsedResponse.projects = projectsResult.projects || [];
      }

      return parsedResponse;
    } catch (error) {
      if (error instanceof OpenAI.APIError) {
        switch (error.status) {
          case 400:
            return {
              valid: false,
              validation_message:
                'Invalid request. Please check the input text.',
            };
          case 401:
            return {
              valid: false,
              validation_message: 'Unauthorized. Please check your API key.',
            };

          case 403:
            return {
              valid: false,
              validation_message:
                'Forbidden. Please check your API key permissions.',
            };
          case 429:
            return {
              valid: false,
              validation_message:
                'Rate limit exceeded. Please wait and retry later.',
            };

          case 500:
            return {
              valid: false,
              validation_message:
                'Internal server error. Please try again later.',
            };
          default:
            return { valid: false, validation_message: 'Error processing CV' };
        }
      }
      this.logger.error(error);
      return { valid: false, validation_message: 'Error processing CV' };
    }
  }

  // Existing method for keyword extraction
  async extractKeywords(text: string): Promise<any> {
    try {
      const response = await this.client.chat.completions.create({
        model: 'gpt-4o-mini',
        messages: [
          {
            role: 'system',
            content:
              'You are an AI specialized in keyword extraction from resumes. Return ONLY a valid JSON object with keywords.',
          },
          {
            role: 'user',
            content: `
              Extract the most important keywords from this CV text. Focus on skills, technologies, domains, methodologies, and industry-specific terms.

              For each keyword:
              1. Assign a relevance score (0.0 to 1.0) based on how central it is to the candidate's profile
              2. Only include significant keywords (score > 0.5)
              3. Return at least 5-10 keywords if possible

              Return ONLY a JSON object in this format:
              {"keywords": [{"keyword": "Keyword1", "relevance_score": 0.9}, {"keyword": "Keyword2", "relevance_score": 0.8}]}

              CV text:
              "${text}"
            `,
          },
        ],
        temperature: 0.3,
      });

      const rawContent = response.choices[0].message.content;
      const jsonValidation = this.isValidJson(rawContent);

      if (!jsonValidation.valid) {
        this.logger.error(
          'Invalid JSON response from keyword extraction',
          rawContent,
        );
        return { keywords: [] };
      }

      return JSON.parse(jsonValidation.cleaned || rawContent);
    } catch (error) {
      this.logger.error('Error extracting keywords', error);
      return { keywords: [] };
    }
  }

  // New method to extract projects as a fallback
  async extractProjects(text: string): Promise<any> {
    try {
      const response = await this.client.chat.completions.create({
        model: 'gpt-4o-mini',
        messages: [
          {
            role: 'system',
            content:
              'You are an AI specialized in extracting project information from resumes. Return ONLY a valid JSON object with projects.',
          },
          {
            role: 'user',
            content: `
              Extract all project information from this CV text. Projects can be mentioned explicitly in "Project" sections,
              or implied within work experience descriptions, technical experience, or portfolio sections.

              For each project identified:
              1. Extract as much detail as possible (title, description, technologies used, dates, impact)
              2. Infer project roles when not explicitly stated
              3. Include URLs if present

              Return ONLY a JSON object in this format:
              {
                "projects": [
                  {
                    "title": "Project title",
                    "description": "Short description (max 200 characters)",
                    "role": "Role in the project",
                    "project_url": "Project URL if available",
                    "impact": "Project impact (max 200 characters)",
                    "technologies_used": ["Technology 1", "Technology 2"],
                    "start_date": "YYYY-MM-DD or null if unavailable",
                    "end_date": "YYYY-MM-DD or '9999-12-31' if ongoing, null if unavailable",
                    "client_or_employer": "Client or employer name if applicable"
                  }
                ]
              }

              CV text:
              "${text}"
            `,
          },
        ],
        temperature: 0.3,
      });

      const rawContent = response.choices[0].message.content;
      const jsonValidation = this.isValidJson(rawContent);

      if (!jsonValidation.valid) {
        this.logger.error(
          'Invalid JSON response from project extraction',
          rawContent,
        );
        return { projects: [] };
      }

      return JSON.parse(jsonValidation.cleaned || rawContent);
    } catch (error) {
      this.logger.error('Error extracting projects', error);
      return { projects: [] };
    }
  }

}
